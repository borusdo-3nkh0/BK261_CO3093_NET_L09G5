%% DESC: Typography, heading format, header/footer, TOC, lists
%% ============================================================

%% --- Packages ---

\usepackage{setspace}                   % line spacing: \setstretch, \onehalfspacing
\usepackage{fancyhdr}                   % custom header/footer
\usepackage{lastpage}                   % \pageref{LastPage} — total page count
\usepackage{titlesec}                   % heading format: \titleformat, \titlespacing
\usepackage{tocloft}                    % TOC appearance control
\usepackage{enumitem}                   % list customization: label, indent, spacing
\usepackage{indentfirst}                % indent first paragraph of each section
\usepackage{ragged2e}                   % improved justification control

%% --- Line Spacing and Paragraph ---

\setstretch{1.5}                        % 1.5 line spacing
\setlength{\parindent}{1cm}             % first-line indent
\setlength{\parskip}{6pt}               % vertical space between paragraphs

%% --- Heading Format ---

% Chapter (Heading 0): centered, all-caps, bold, starts new page
\titleformat{\chapter}[display]
    {\normalfont\Huge\bfseries\centering}
    {\MakeUppercase{\chaptername}\ \thechapter}
    {0.5em}
    {\MakeUppercase}

\titlespacing{\chapter}{0pt}{0pt}{2em}

% Section (Heading 1): left, bold, sentence case
\renewcommand{\thesection}{\arabic{section}}
\titleformat{\section}
    {\normalfont\large\bfseries}
    {\thesection.}
    {0.5em}
    {}

\titlespacing{\section}{0pt}{2em}{1em}

% Subsection (Heading 2): left, bold, normal size
\titleformat{\subsection}
    {\normalfont\normalsize\bfseries}
    {\thesubsection.}
    {0.5em}
    {}

\titlespacing{\subsection}{0pt}{1.5em}{0.75em}

% Subsubsection (Heading 3): left, italic, normal size
\titleformat{\subsubsection}
    {\normalfont\normalsize\bfseries\itshape}
    {\thesubsubsection.}
    {0.5em}
    {}

\titlespacing{\subsubsection}{0pt}{1em}{0em}

%% --- Header and Footer ---

\pagestyle{fancy}
\setlength{\headwidth}{\textwidth}
\fancyhf{}                              % clear all defaults

\fancyhead[L]{\docCourse \, (\docCourseCode)}
% \fancyhead[C]{\docTitle}              % uncomment to show document title in center
\fancyhead[R]{\textcolor{colorTextSoft}{\thepage/\pageref*{LastPage}}}

\renewcommand{\headrulewidth}{0.5pt}
\renewcommand{\headrule}{{\color{colorPrimary}\hrule width\headwidth height\headrulewidth \vskip-\headrulewidth}}
\renewcommand{\footrulewidth}{0.5pt}
\renewcommand{\footrule}{{\color{colorPrimary}\hrule width\headwidth height\footrulewidth \vskip-\footrulewidth}}

% Plain style — used on first page of each chapter
\fancypagestyle{plain}{
    \fancyhf{}
    \fancyfoot[C]{\thepage}             % page number centered in footer
    \renewcommand{\headrulewidth}{0pt}
    \renewcommand{\footrulewidth}{0pt}
}

%% --- Table of Contents ---

% TOC title: match \section style
\renewcommand{\cfttoctitlefont}{\large\bfseries}

% Entry typography
\renewcommand{\cftsecfont}{\bfseries}
\renewcommand{\cftsecpagefont}{\bfseries}
\renewcommand{\cftsubsecfont}{\normalfont}
\renewcommand{\cftsubsubsecfont}{\normalfont\small}

% Vertical spacing before entries
\setlength{\cftbeforesecskip}{0.5em}
\setlength{\cftbeforesubsecskip}{0.25em}

% Page number column
\cftsetpnumwidth{1.5em}
\cftsetrmarg{2.5em}

% Dot leader
\renewcommand{\cftsecleader}{\cftdotfill{\cftdotsep}}
\renewcommand{\cftdotsep}{3}

% Indentation
\cftsetindents{section}{0em}{2em}
\cftsetindents{subsection}{2em}{2.5em}
\cftsetindents{subsubsection}{4.5em}{3em}

%% --- Lists ---

% Itemize
\setlist[itemize,1]{label=\textbullet,  itemindent=14pt}
\setlist[itemize,2]{label=\textendash,  itemindent=14pt}
\setlist[itemize,3]{label=$\ast$,       itemindent=14pt}

% Enumerate
\setlist[enumerate,1]{label=(\arabic*), itemindent=14pt}
\setlist[enumerate,2]{label=(\alph*),   itemindent=14pt}
\setlist[enumerate,3]{label=(\roman*),  itemindent=14pt}

% Reduce spacing between items
\setlist{noitemsep, topsep=0.5em, parsep=0pt}
