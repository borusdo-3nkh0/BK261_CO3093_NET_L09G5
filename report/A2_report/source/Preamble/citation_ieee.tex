%% ============================================================
%% FILE: citation_ieee.tex
%% DESC: IEEE numeric citation style via biblatex
%% ============================================================

%% --- Packages ---

\usepackage[
    autostyle = false,
    style     = english,
]{csquotes}                             % smart quotes; required by biblatex

\usepackage[
    backend     = biber,               % NOTE: requires biber, not bibtex
    style       = ieee,                % IEEE numeric style
    citestyle   = numeric-comp,        % compress ranges: [1,2,3] → [1–3]
    sorting     = none,                % citation order = appearance order in text
    dashed      = false,               % repeat author name instead of dash
    doi         = true,
    url         = true,
    isbn        = false,
    eprint      = false,
    maxbibnames = 6,                   % show up to 6 authors before et al.
    minbibnames = 1,
]{biblatex}

%% --- Language Mapping ---

% suppress "Retrieved from" noise
\DeclareLanguageMapping{vietnamese}{vietnamese-ieee}

%% --- URL Line Breaking ---

\setcounter{biburllcpenalty}{7000}
\setcounter{biburlucpenalty}{8000}
\setcounter{biburlnumpenalty}{9000}

%% --- Bibliography Heading ---

% matches \section* style from styling.tex
\defbibheading{bibliography}[\bibname]{%
    \section*{#1}%
    \addcontentsline{toc}{section}{#1}%
    \markboth{#1}{#1}%
}

%% --- Citation Aliases ---

% USAGE:
%    \citep{key}   →  [n]           standard numeric
%    \citet{key}   →  Author [n]    author-prominent
\newcommand{\citep}[1]{\cite{#1}}
\newcommand{\citet}[1]{\citeauthor{#1}~\cite{#1}}

%% --- Bibliography Appearance ---

\renewcommand*{\bibfont}{\small}

%% ============================================================
%% HOW TO USE
%% ============================================================
%%  In main.tex:
%%
%%    \addbibresource{refs.bib}          % in preamble
%%
%%    \cite{key}                         % → [1]
%%    \citep{key}                        % → [1]   (alias)
%%    \citet{key}                        % → Author [1]
%%    \parencite{key}                    % → [1]   (same as \cite for numeric)
%%    \footcite{key}                     % → footnote citation
%%
%%    \printbibliography
%%    \printbibliography[title={References}]
%%
%%  Compile order:
%%    pdflatex → biber → pdflatex → pdflatex
%%
%%  Common entry types in refs.bib:
%%    @article, @inproceedings, @book,
%%    @online (use urldate field), @techreport
%% ============================================================
